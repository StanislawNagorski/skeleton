**Step 1: Rewrite counter data source (`lib/features/home_page/data/local_database/counter_local_database.dart`).**

Replace entire file with Drift-backed implementation per D-08, D-09. The data source returns Drift's generated `CounterEntry` type (not `CounterModel`) -- the repository handles mapping.

```dart
import 'package:drift/drift.dart';
import 'package:injectable/injectable.dart';

import '../../../../core/database/app_database.dart';

/// Data source for counter persistence.
///
/// Queries Drift [AppDatabase] directly. Returns Drift-generated
/// [CounterEntry] types -- mapping to domain [CounterModel] happens
/// in the repository layer.
abstract class CounterLocalDataBase {
  /// Returns the first counter entry, or `null` if none exists.
  Future<CounterEntry?> getCounter();

  /// Upserts a counter entry with the given [value].
  Future<void> setCounter({required int value});

  /// Watches the first counter entry reactively.
  Stream<CounterEntry?> watchCounter();
}

/// Drift-backed implementation of [CounterLocalDataBase].
@LazySingleton(as: CounterLocalDataBase)
class CounterLocalDataBaseImpl implements CounterLocalDataBase {
  /// Creates a [CounterLocalDataBaseImpl] with the injected [AppDatabase].
  CounterLocalDataBaseImpl(this._db);

  final AppDatabase _db;

  @override
  Future<CounterEntry?> getCounter() async {
    return (_db.select(_db.counterEntries)..limit(1)).getSingleOrNull();
  }

  @override
  Future<void> setCounter({required int value}) async {
    await _db.into(_db.counterEntries).insertOnConflictUpdate(
          CounterEntriesCompanion.insert(value: Value(value)),
        );
  }

  @override
  Stream<CounterEntry?> watchCounter() {
    return (_db.select(_db.counterEntries)..limit(1)).watchSingleOrNull();
  }
}
```

Note: `CounterEntry` and `CounterEntriesCompanion` are generated types from `app_database.g.dart`. They will exist after build_runner runs in Task 3.

```dart
import 'package:injectable/injectable.dart';
import 'package:logger/logger.dart';

import '../../../../core/errors/failures.dart';
import '../../../../core/errors/result.dart';
import '../local_database/counter_local_database.dart';
import '../model/counter_model.dart';

/// Handles errors from counter data sources and returns typed [Result]s.
abstract class CounterRepository {
  /// Reads the current counter value.
  Future<Result<CounterModel>> getCounter();

  /// Persists the counter [value].
  Future<Result<void>> setCounter({required CounterModel counter});

  /// Watches counter changes as a reactive stream of [Result]s.
  Stream<Result<CounterModel>> watchCounter();
}

/// Implementation backed by [CounterLocalDataBase].
@LazySingleton(as: CounterRepository)
class CounterRepositoryImpl implements CounterRepository {
  /// Creates a [CounterRepositoryImpl] with the given data source.
  CounterRepositoryImpl(this._localDataBase);

  final CounterLocalDataBase _localDataBase;
  final Logger _logger = Logger();

  @override
  Future<Result<CounterModel>> getCounter() async {
    try {
      final entry = await _localDataBase.getCounter();
      if (entry == null) {
        return const Failure(LocalDataNotFound('Counter not found'));
      }
      return Success(
        CounterModel(id: entry.id, value: entry.value),
      );
    } catch (e) {
      _logger.e('Local data source exception: $e');
      return Failure(LocalDataSource(e.toString()));
    }
  }

  @override
  Future<Result<void>> setCounter({required CounterModel counter}) async {
    try {
      await _localDataBase.setCounter(value: counter.value);
      return const Success(null);
    } catch (e) {
      _logger.e('Local data source exception: $e');
      return Failure(LocalDataSource(e.toString()));
    }
  }

  @override
  Stream<Result<CounterModel>> watchCounter() {
    return _localDataBase.watchCounter().map((entry) {
      try {
        if (entry == null) {
          return const Failure<CounterModel>(
            LocalDataNotFound('Counter not found'),
          );
        }
        return Success(
          CounterModel(id: entry.id, value: entry.value),
        );
      } catch (e) {
        _logger.e('Local data source exception: $e');
        return Failure<CounterModel>(LocalDataSource(e.toString()));
      }
    });
  }
}
```

```dart
import 'dart:async';

import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:injectable/injectable.dart';

import '../../../../core/errors/result.dart';
import '../../data/model/counter_model.dart';
import '../../domain/counter_service.dart';

part 'home_state.dart';

/// Manages counter page state.
@injectable
class HomeCubit extends Cubit<HomeState> {
  /// Creates a [HomeCubit] backed by the given [CounterService].
  HomeCubit(this._counterService) : super(HomeState.initial());

  final CounterService _counterService;
  StreamSubscription<Result<CounterModel>>? _counterStream;

  /// Initializes the counter -- creates a default if none exists,
  /// then subscribes to reactive updates.
  Future<void> init() async {
    final counter = await _getCounter();
    if (counter == null) {
      await _counterService.setCounter(
        counter: const CounterModel(value: 0),
      );
    }

    _counterStream = _counterService.watchCounter().listen(
      (result) {
        switch (result) {
          case Success(:final value):
            emit(
              state.copyWith(
                status: HomePageStatus.success,
                counter: value,
              ),
            );
          case Failure():
            emit(state.copyWith(status: HomePageStatus.error));
        }
      },
    );
  }

  /// Increments the counter by 1.
  void increment() {
    final updatedData = state.counter.copyWith(
      value: state.counter.value + 1,
    );
    unawaited(
      _counterService.setCounter(counter: updatedData),
    );
  }

  /// Decrements the counter by 1.
  void decrement() {
    final updatedData = state.counter.copyWith(
      value: state.counter.value - 1,
    );
    unawaited(
      _counterService.setCounter(counter: updatedData),
    );
  }

  Future<CounterModel?> _getCounter() async {
    final result = await _counterService.getCounter();
    return result.valueOrNull;
  }

  @override
  Future<void> close() {
    _counterStream?.cancel();
    return super.close();
  }
}
```

Note: `unawaited` is from `dart:async` (already imported). This satisfies the `unawaited_futures` lint rule that Plan 01 promoted to error level.

```dart
/// Abstract secure storage service for encrypted key-value persistence.
///
/// Use for token-class data: auth tokens, refresh tokens, API keys.
/// Do NOT use for non-sensitive preferences -- use [PreferencesService]
/// instead.
abstract class SecureStorageService {
  /// Reads the encrypted value for [key], or `null` if not set.
  Future<String?> read({required String key});

  /// Writes an encrypted [value] for [key].
  Future<void> write({required String key, required String value});

  /// Deletes the encrypted value for [key].
  Future<void> delete({required String key});

  /// Deletes all stored encrypted values.
  Future<void> deleteAll();
}
```

```dart
/// [FlutterSecureStorage]-backed implementation of [SecureStorageService].
///
/// Android: uses `EncryptedSharedPreferences` for hardware-backed encryption.
/// iOS: uses Keychain with `first_unlock_this_device` accessibility.
@LazySingleton(as: SecureStorageService)
class SecureStorageServiceImpl implements SecureStorageService {
  static const _storage = FlutterSecureStorage(
    aOptions: AndroidOptions(encryptedSharedPreferences: true),
    iOptions: IOSOptions(

    ),
  );

  Future<String?> read({required String key}) =>
      _storage.read(key: key);

  @override
  Future<void> write({required String key, required String value}) =>
      _storage.write(key: key, value: value);

  @override
  Future<void> delete({required String key}) =>
      _storage.delete(key: key);

  @override
  Future<void> deleteAll() => _storage.deleteAll();
}
```
