```dart
// ignore_for_file: discarded_futures

import 'dart:isolate';

import 'package:firebase_app_check/firebase_app_check.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:firebase_crashlytics/firebase_crashlytics.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/widgets.dart';
import 'package:logger/logger.dart';

import 'package:skeleton/core/injectable/injection.dart';
import 'package:skeleton/firebase_options.dart';

/// Bootstraps the app: Firebase init (with dev fallback), error
/// reporting, App Check, and dependency injection.
abstract class StartUpService {
  static final Logger _logger = Logger();

  /// Full startup sequence per D-03:
  /// 1. WidgetsFlutterBinding.ensureInitialized() — called by main()
  /// 2. Try Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform)
  /// 3. If success: env='prod', wire Crashlytics + App Check
  /// 4. If failure: env='dev' (FakeAuthService, no Firebase deps)
  /// 5. configureDependenciesApp(environment: env)
  static Future<void> initServices() async {
    String environment = 'dev';

    try {
      await Firebase.initializeApp(
        options: DefaultFirebaseOptions.currentPlatform,
      );
      environment = 'prod';
      _configureCrashlytics();
      await _configureAppCheck();
    } on Exception catch (e) {
      // No Firebase config files — running in dev mode (D-01).
      // FakeAuthService will be registered; no Firebase services in DI.
      _logger.w('Firebase not configured: $e. Running in dev mode.');
    }

    await configureDependenciesApp(environment: environment);
  }

  /// Wires the Crashlytics error handler trio per D-07/D-08.
  ///
  /// - Handler 1: Flutter framework errors via FlutterError.onError
  /// - Handler 2: Async Dart errors via PlatformDispatcher.instance.onError
  /// - Handler 3: Spawned isolate errors via Isolate.current.addErrorListener
  ///
  /// Collection disabled in debug (D-07): debug crashes stay in console only.
  static void _configureCrashlytics() {
    // D-07: No dashboard noise from dev sessions.
    FirebaseCrashlytics.instance
        .setCrashlyticsCollectionEnabled(kReleaseMode);

    // Handler 1: Flutter framework errors (synchronous).
    FlutterError.onError =
        FirebaseCrashlytics.instance.recordFlutterFatalError;

    // Handler 2: Async Dart errors not caught by the Flutter framework.
    PlatformDispatcher.instance.onError = (error, stack) {
      FirebaseCrashlytics.instance
          .recordError(error, stack, fatal: true);
      return true;
    };

    // Handler 3: Errors in spawned isolates (D-08).
    // Catches errors from downstream app isolates (image processing,
    // Drift heavy queries, etc.).
    Isolate.current.addErrorListener(
      RawReceivePort((dynamic pair) async {
        final errorAndStack = pair as List<dynamic>;
        await FirebaseCrashlytics.instance.recordError(
          errorAndStack[0],
          StackTrace.fromString(errorAndStack[1] as String),
          fatal: true,
        );
      }).sendPort,
    );
  }

  /// App Check release-only activation per D-09/D-10.
  ///
  /// Dev builds skip entirely — no debug provider, no debug-token
  /// registration friction. Firebase endpoints are unprotected in dev.
  static Future<void> _configureAppCheck() async {
    if (!kReleaseMode) return; // D-10: Skip entirely in dev.

    await FirebaseAppCheck.instance.activate(
      androidProvider: AndroidProvider.playIntegrity,
      appleProvider: AppleProvider.appAttest, // D-09: appAttest, not deviceCheck.
    );
  }
}
```

Create a new directory `lib/core/firebase/` and file `lib/core/firebase/firebase_module.dart`. This injectable module registers Firebase service instances as lazy singletons in the prod environment. Pattern follows the existing `DatabaseModule` in `lib/core/database/database_module.dart`.

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_analytics/firebase_analytics.dart';
import 'package:firebase_performance/firebase_performance.dart';
import 'package:injectable/injectable.dart';

/// Registers Firebase service instances in the `'prod'` DI environment.
///
/// These are only available when Firebase is initialized (i.e. when
/// `flutterfire configure` has been run and config files are present).
/// The `'dev'` environment does not register these — accessing them
/// without Firebase would throw.
@module
abstract class FirebaseModule {
  /// Firebase Analytics instance for event tracking.
  ///
  /// No events are wired in the skeleton — analytics taxonomy
  /// is per-app. Registered so downstream features can inject it.
  @lazySingleton
  @Environment(Environment.prod)
  FirebaseAnalytics get analytics => FirebaseAnalytics.instance;

  /// Firebase Performance automatic tracing.
  ///
  /// Automatic tracing is enabled by default — no manual traces
  /// are required for the skeleton counter example.
  @lazySingleton
  @Environment(Environment.prod)
  FirebasePerformance get performance => FirebasePerformance.instance;

  /// Cloud Firestore with offline persistence enabled.
  ///
  /// Persistence is ON by default on mobile. Explicit settings
  /// for clarity and to document the skeleton's stance.
  @lazySingleton
  @Environment(Environment.prod)
  FirebaseFirestore get firestore {
    final instance = FirebaseFirestore.instance;
    instance.settings = const Settings(
      persistenceEnabled: true,
    );
    return instance;
  }
}
```
