```dart
final fileLoader = FileLoader.fromAsset('assets/rive/demo.riv', riveFactory: Factory.rive);

  fileLoader: fileLoader,
  stateMachineSelector: const StateMachineDefault(),
  builder: (context, state) => switch (state) {
    RiveLoading() => const Center(child: CircularProgressIndicator()),
    RiveFailed() => Center(child: Text('Failed: ${state.error}')),
```

   ```dart
   import 'package:injectable/injectable.dart';

   part 'rive_demo_state.dart';

   /// Manages RIVE demo page toggle state.
   ///
   /// The cubit tracks the boolean toggle value. The actual RIVE
   /// state-machine input is managed in the page's StatefulWidget
   /// because RIVE controllers are tied to the render tree and must
   /// be disposed in [State.dispose], not in the cubit.
   @injectable
   class RiveDemoCubit extends Cubit<RiveDemoState> {
     RiveDemoCubit() : super(RiveDemoState.initial());

     /// Toggles the animation state-machine boolean input.
     void toggleAnimation() {
       emit(state.copyWith(isToggled: !state.isToggled));
     }

     /// Called when the RIVE file loads successfully.
     void onLoaded() {
       emit(state.copyWith(status: RiveDemoPageStatus.loaded));
     }

     /// Called when the RIVE file fails to load.
     void onError() {
       emit(state.copyWith(status: RiveDemoPageStatus.error));
     }
   }
   ```

```dart
import 'package:auto_route/auto_route.dart';
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:rive/rive.dart';

import 'package:skeleton/core/extensions/context_extension.dart';

/// Dedicated RIVE demo page showcasing .riv asset loading,
/// state-machine interaction, and proper cleanup (D-16, D-17).
@RoutePage()
class RiveDemoPage extends StatelessWidget {
  const RiveDemoPage({super.key});

  @override
  Widget build(BuildContext context) => BlocProvider(
        create: (context) => getIt<RiveDemoCubit>(),
        child: const _Page(),
      );
}
```

The private `_Page` is a `StatefulWidget` that owns the RIVE resources:

```dart
class _Page extends StatefulWidget {
  const _Page();
  @override
  State<_Page> createState() => _PageState();
}

class _PageState extends State<_Page> {
  late final FileLoader _fileLoader;
  BooleanInput? _toggleInput;

  @override
  void initState() {
    super.initState();
    _fileLoader = FileLoader.fromAsset(
      'assets/rive/demo.riv',
      riveFactory: Factory.rive,
    );
  }

  @override
  void dispose() {
    // Dispose in reverse order of creation per RESEARCH.md Pitfall 5.
    _toggleInput?.dispose();
    _fileLoader.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Scaffold(
        appBar: AppBar(
          backgroundColor: context.theme.colorScheme.inversePrimary,
          title: Text(context.localizations.riveDemo_title),
        ),
        body: Padding(
          padding: const EdgeInsets.all(AppDimensions.spacingMedium),
          child: Column(
            children: [
              // Animation canvas in a Card (per UI-SPEC: 300px height, 12px radius)
              Card(
                shape: RoundedRectangleBorder(
                  borderRadius: BorderRadius.circular(
                    AppDimensions.radiusMedium,
                  ),
                ),
                clipBehavior: Clip.antiAlias,
                child: SizedBox(
                  width: double.infinity,
                  height: 300,
                  child: RiveWidgetBuilder(
                    fileLoader: _fileLoader,
                    stateMachineSelector: const StateMachineDefault(),
                    builder: (context, state) => switch (state) {
                      RiveLoading() => const Center(
                          child: CircularProgressIndicator(),
                        ),
                      RiveFailed() => Center(
                          child: Text(
                            'Failed to load animation: ${state.error}',
                            style: context.theme.textTheme.bodyMedium,
                          ),
                        ),
                      RiveLoaded() => _buildLoadedAnimation(
                          context,
                          state.controller,
                        ),
                    },
                  ),
                ),
              ),
              const SizedBox(height: AppDimensions.spacingMedium),
              // State-machine toggle (per D-17)
              BlocBuilder<RiveDemoCubit, RiveDemoState>(
                builder: (context, state) => SwitchListTile(
                  title: Text(
                    context.localizations.riveDemo_toggleLabel,
                    style: context.theme.textTheme.bodyMedium,
                  ),
                  value: state.isToggled,
                  onChanged: (value) {
                    context.read<RiveDemoCubit>().toggleAnimation();
                    _toggleInput?.value = value;
                  },
                ),
              ),
              const SizedBox(height: AppDimensions.spacingBase),
              // Description text
              Padding(
                padding: const EdgeInsets.symmetric(
                  horizontal: AppDimensions.spacingMedium,
                ),
                child: Text(
                  context.localizations.riveDemo_description,
                  style: context.theme.textTheme.bodyMedium?.copyWith(
                    color: context.theme.colorScheme.onSurfaceVariant,
                  ),
                ),
              ),
            ],
          ),
        ),
      );

  Widget _buildLoadedAnimation(
    BuildContext context,
    RiveWidgetController controller,
  ) {
    // Acquire the boolean input handle on first load.
    _toggleInput ??= controller.stateMachine.boolean('Toggle');
    // Sync the toggle input with cubit state.
    final isToggled = context.read<RiveDemoCubit>().state.isToggled;
    _toggleInput?.value = isToggled;
    return RiveWidget(
      controller: controller,
      fit: Fit.contain,
    );
  }
}
```
