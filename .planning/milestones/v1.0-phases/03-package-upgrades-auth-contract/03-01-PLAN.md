import 'package:skeleton/core/auth/auth_user.dart';
import 'package:skeleton/core/errors/failures.dart';

/// Sealed result type for authentication operations.
///
/// Unlike [Result<T>], auth operations have a third state — cancellation
/// (e.g. user dismisses the Google/Apple sign-in sheet).
///
/// Use exhaustive `switch` pattern matching to consume:
/// ```dart
/// switch (result) {
///   case AuthSuccess(:final user): // handle success
///   case AuthFailure(:final failure): // handle failure
///   case AuthCancelled(): // handle cancellation
/// }
/// ```
sealed class AuthResult {
  const AuthResult();
}

/// Authentication succeeded, carrying the authenticated [user].
final class AuthSuccess extends AuthResult {
  const AuthSuccess(this.user);
  final AuthUser user;
}

/// Authentication failed with an [AppFailure].
final class AuthFailure extends AuthResult {
  const AuthFailure(this.failure);
  final AppFailure failure;
}

/// Authentication was cancelled by the user.
final class AuthCancelled extends AuthResult {
  const AuthCancelled();
}
```

```dart
import 'package:skeleton/core/auth/auth_result.dart';
import 'package:skeleton/core/auth/auth_user.dart';

/// Central authentication boundary.
///
/// Exposes a reactive [authStateChanges] stream and request-response methods
/// for every supported sign-in provider.  Implementations:
/// - [FakeAuthService] — in-memory, ships with the skeleton (Phase 3)
/// - `FirebaseAuthService` — real Firebase Auth (Phase 5)
abstract class AuthService {
  /// Emits the current [AuthUser] on sign-in, `null` on sign-out.
  Stream<AuthUser?> get authStateChanges;

  /// The currently signed-in user, or `null` if signed out.
  AuthUser? get currentUser;

  /// Signs in with email + password.
  Future<AuthResult> signInWithEmail({
    required String email,
    required String password,
  });

  /// Creates a new account with email + password.
  Future<AuthResult> signUpWithEmail({
    required String email,
    required String password,
  });

  /// Initiates Google Sign-In flow.
  Future<AuthResult> signInWithGoogle();

  /// Initiates Apple Sign-In flow.
  Future<AuthResult> signInWithApple();

  /// Signs out of all active providers.
  Future<void> signOut();

  /// Sends a password-reset email.
  Future<AuthResult> sendPasswordReset({required String email});
}
```

**lib/core/auth/fake_auth_service.dart** (NEW file, per D-06, D-07):
```dart
import 'dart:async';

import 'package:injectable/injectable.dart';

import 'package:skeleton/core/auth/auth_result.dart';
import 'package:skeleton/core/auth/auth_service.dart';
import 'package:skeleton/core/auth/auth_user.dart';
import 'package:skeleton/core/errors/failures.dart';

/// In-memory [AuthService] for development without Firebase.
///
/// Signs in any email/password combination, returns a deterministic
/// [AuthUser] with `uid` derived from the email.  Kept permanently as a
/// test double — Phase 5 adds `@Environment` switching.
@LazySingleton(as: AuthService)
class FakeAuthService implements AuthService {
  final StreamController<AuthUser?> _controller =
      StreamController<AuthUser?>.broadcast();

  AuthUser? _currentUser;

  @override
  Stream<AuthUser?> get authStateChanges => _controller.stream;

  @override
  AuthUser? get currentUser => _currentUser;

  @override
  Future<AuthResult> signInWithEmail({
    required String email,
    required String password,
  }) async {
    final user = AuthUser(
      uid: 'fake-${email.hashCode}',
      email: email,
      displayName: null,
      photoUrl: null,
      emailVerified: false,
    );
    _currentUser = user;
    _controller.add(user);
    return AuthSuccess(user);
  }

  @override
  Future<AuthResult> signUpWithEmail({
    required String email,
    required String password,
  }) async {
    final user = AuthUser(
      uid: 'fake-${email.hashCode}',
      email: email,
      displayName: null,
      photoUrl: null,
      emailVerified: false,
    );
    _currentUser = user;
    _controller.add(user);
    return AuthSuccess(user);
  }

  @override
  Future<AuthResult> signInWithGoogle() async {
    return const AuthCancelled();
  }

  @override
  Future<AuthResult> signInWithApple() async {
    return const AuthCancelled();
  }

  @override
  Future<void> signOut() async {
    _currentUser = null;
    _controller.add(null);
  }

  @override
  Future<AuthResult> sendPasswordReset({required String email}) async {
    return AuthSuccess(
      AuthUser(
        uid: 'fake-${email.hashCode}',
        email: email,
        displayName: null,
        photoUrl: null,
        emailVerified: false,
      ),
    );
  }

  /// Releases the internal [StreamController].
  ///
  /// Called by `get_it.disposeDependencies()` during test teardown.
  void dispose() {
    _controller.close();
  }
}
```
