**Implementation structure:**

```dart
import 'dart:async';
import 'dart:convert';

import 'package:crypto/crypto.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:google_sign_in/google_sign_in.dart';
import 'package:injectable/injectable.dart';
import 'package:sign_in_with_apple/sign_in_with_apple.dart';

import 'package:skeleton/core/auth/auth_result.dart';
import 'package:skeleton/core/auth/auth_service.dart';
import 'package:skeleton/core/auth/auth_user.dart';
import 'package:skeleton/core/errors/failures.dart';

/// Production [AuthService] backed by Firebase Auth.
///
/// Registered in the `'prod'` DI environment — only available when
/// `Firebase.initializeApp` succeeds (i.e. `flutterfire configure`
/// has been run).
@LazySingleton(as: AuthService, env: [Environment.prod])
class FirebaseAuthService implements AuthService {
  FirebaseAuthService() {
    _subscription = FirebaseAuth.instance
        .authStateChanges()
        .map(_mapUser)
        .listen(_controller.add);
  }

  final StreamController<AuthUser?> _controller =
      StreamController<AuthUser?>.broadcast();
  late final StreamSubscription<AuthUser?> _subscription;

  @override
  Stream<AuthUser?> get authStateChanges => _controller.stream;

  @override
  AuthUser? get currentUser => _mapUser(FirebaseAuth.instance.currentUser);

  // -- Email + Password (AUTH-02) --

  @override
  Future<AuthResult> signInWithEmail({
    required String email,
    required String password,
  }) async {
    try {
      final credential = await FirebaseAuth.instance
          .signInWithEmailAndPassword(email: email, password: password);
      return AuthSuccess(_mapUser(credential.user)!);
    } on FirebaseAuthException catch (e) {
      return AuthFailure(AuthError(_mapAuthErrorCode(e.code)));
    } on Exception catch (e) {
      return AuthFailure(AuthError(e.toString()));
    }
  }

  @override
  Future<AuthResult> signUpWithEmail({
    required String email,
    required String password,
  }) async {
    try {
      final credential = await FirebaseAuth.instance
          .createUserWithEmailAndPassword(email: email, password: password);
      return AuthSuccess(_mapUser(credential.user)!);
    } on FirebaseAuthException catch (e) {
      return AuthFailure(AuthError(_mapAuthErrorCode(e.code)));
    } on Exception catch (e) {
      return AuthFailure(AuthError(e.toString()));
    }
  }

  // -- Google Sign-In (AUTH-03) --
  // EXECUTOR: Verify google_sign_in 7.x API via Context7 before implementing.
  // RESEARCH.md A1/A2 flag the exact API as medium/high risk.
  // The pattern below is from RESEARCH.md but may need adaptation.

  @override
  Future<AuthResult> signInWithGoogle() async {
    // TODO(executor): Look up google_sign_in 7.x actual API.
    // The implementation below follows RESEARCH.md Pattern 4.
    // Adapt if the real API differs.
    try {
      final googleSignIn = GoogleSignIn.instance;
      // serverClientId: The Web OAuth client ID from Google Cloud Console.
      // Per D-05, this should come from DefaultFirebaseOptions if possible.
      // If FirebaseOptions doesn't have a webClientId field, the
      // developer must pass it post-bootstrap.
      await googleSignIn.initialize(
        serverClientId: _webClientId,
      );
      final googleUser = await googleSignIn.authenticate();
      // Token retrieval — adapt to actual 7.x API:
      final googleAuth = await googleUser.authorizationClient
          .authorizationForScopes([]);
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth?.accessToken,
        idToken: googleAuth?.idToken,
      );
      final userCredential = await FirebaseAuth.instance
          .signInWithCredential(credential);
      return AuthSuccess(_mapUser(userCredential.user)!);
    } on Exception catch (e) {
      if (_isUserCancellation(e)) return const AuthCancelled();
      return AuthFailure(AuthError(e.toString()));
    }
  }

  // -- Apple Sign-In (AUTH-04) --

  @override
  Future<AuthResult> signInWithApple() async {
    try {
      final rawNonce = generateNonce(); // from firebase_auth (A3)
      final nonce = sha256.convert(utf8.encode(rawNonce)).toString();

      final appleCredential = await SignInWithApple.getAppleIDCredential(
        scopes: [
          AppleIDAuthorizationScopes.email,
          AppleIDAuthorizationScopes.fullName,
        ],
        nonce: nonce,
      );

      final oauthCredential = OAuthProvider('apple.com').credential(
        idToken: appleCredential.identityToken,
        rawNonce: rawNonce,
      );

      final userCredential = await FirebaseAuth.instance
          .signInWithCredential(oauthCredential);
      return AuthSuccess(_mapUser(userCredential.user)!);
    } on SignInWithAppleAuthorizationException catch (e) {
      if (e.code == AuthorizationErrorCode.canceled) {
        return const AuthCancelled();
      }
      return AuthFailure(AuthError(e.toString()));
    } on Exception catch (e) {
      return AuthFailure(AuthError(e.toString()));
    }
  }

  // -- Sign Out (AUTH-06) --

  @override
  Future<void> signOut() async {
    await GoogleSignIn.instance.signOut();
    await FirebaseAuth.instance.signOut();
    // authStateChanges emits null via the Firebase stream subscription
  }

  // -- Password Reset --

  @override
  Future<AuthResult> sendPasswordReset({required String email}) async {
    try {
      await FirebaseAuth.instance.sendPasswordResetEmail(email: email);
      // Return a success with a minimal AuthUser (uid derived from email).
      // The actual user may or may not exist — Firebase sends the email
      // regardless (to prevent user enumeration).
      return AuthSuccess(AuthUser(
        uid: email,
        email: email,
        displayName: null,
        photoUrl: null,
        emailVerified: false,
      ));
    } on FirebaseAuthException catch (e) {
      return AuthFailure(AuthError(_mapAuthErrorCode(e.code)));
    } on Exception catch (e) {
      return AuthFailure(AuthError(e.toString()));
    }
  }

  // -- Helpers --

  AuthUser? _mapUser(User? user) {
    if (user == null) return null;
    return AuthUser(
      uid: user.uid,
      email: user.email,
      displayName: user.displayName,
      photoUrl: user.photoURL,
      emailVerified: user.emailVerified,
    );
  }

  /// Maps Firebase error codes to user-readable messages.
  String _mapAuthErrorCode(String code) => switch (code) {
        'user-not-found' => 'No account found with this email.',
        'wrong-password' => 'Incorrect password.',
        'invalid-email' => 'Invalid email address.',
        'user-disabled' => 'This account has been disabled.',
        'email-already-in-use' =>
          'An account already exists with this email.',
        'weak-password' => 'Password is too weak.',
        'too-many-requests' => 'Too many attempts. Try again later.',
        'operation-not-allowed' =>
          'This sign-in method is not enabled.',
        'invalid-credential' => 'Invalid credentials.',
        _ => 'Authentication error: $code',
      };

  bool _isUserCancellation(Exception e) =>
      e.toString().contains('canceled') ||
      e.toString().contains('cancelled');

  /// Web OAuth client ID for Google Sign-In on Android (D-05).
  ///
  /// After `flutterfire configure`, DefaultFirebaseOptions may contain
  /// the web client ID. If not, this must be set per-project.
  /// The executor should check FirebaseOptions fields and adapt.
  String get _webClientId {
    // Attempt to source from DefaultFirebaseOptions if a suitable field exists.
    // Fallback: empty string triggers google_sign_in configuration error,
    // which is caught and returns AuthFailure with a descriptive message.
    // Post-bootstrap, the developer sets this up per the README checklist.
    return '';
  }

  /// Releases resources. Called by get_it during app teardown.
  @disposeMethod
  Future<void> dispose() async {
    await _subscription.cancel();
    await _controller.close();
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User input -> FirebaseAuth | Email/password from UI crosses to Firebase server-side validation |
| Google/Apple OAuth -> FirebaseAuth | Third-party OAuth tokens exchanged for Firebase credentials |
| Firebase -> AuthUser mapping | Firebase User mapped to pure Dart AuthUser (no Firebase types leak) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-07 | Spoofing | Apple Sign-In | mitigate | SHA-256 nonce generated per sign-in attempt via `generateNonce()` + `crypto` package; prevents replay attacks |
| T-04-08 | Spoofing | Google Sign-In | mitigate | SHA-1 fingerprint registration documented in README (AUTH-05); without correct fingerprint, Google rejects the request |
| T-04-09 | Information Disclosure | Firebase error codes | mitigate | `_mapAuthErrorCode` maps codes to generic messages — does not expose internal Firebase error details to UI |
| T-04-10 | Denial of Service | Auth rate limiting | accept | Firebase server-side rate limiting (`too-many-requests` code); skeleton maps the code but enforcement is Firebase's responsibility |
| T-04-11 | Information Disclosure | Apple first-sign-in email | accept | Apple returns email/name only on first sign-in (Pitfall 2); skeleton persists to AuthUser but downstream apps must store to Firestore. Documented in README |
| T-04-12 | Spoofing | Password reset enumeration | mitigate | Firebase sends reset email regardless of account existence (prevents user enumeration); skeleton follows this pattern |
</threat_model>
